let expenses = [5000, 2000, 10000, 1500, 2500]; let total = expenses.reduce((sum, expense) => sum + expense, 0); let average = total / expenses.length; let tax = total * 0.10; total += tax; console.log("Original Expenses:", (total - tax).toFixed(2)); console.log("Average Expense:", average.toFixed(2)); console.log("Tax (10%):", tax.toFixed(2)); console.log("Final Amount After Tax:", total.toFixed(2));
